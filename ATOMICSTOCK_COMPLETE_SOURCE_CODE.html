527/**
528 * Predictive Run-out & Velocity Engine.
529 * Analyzes trailing 14-day outbound consumption to derive burn rate (units/day),
530 * forecast run-out date, and compute EOQ replenishment recommendations.
531 */
532app.get('/api/v1/analytics/predictive-restock', requireAuth, enforceNodeScope, async (req, res): Promise<void> => {
533  try {
534    const warehouseId = req.query.warehouseId as string | undefined;
535    const inventoryWhere = warehouseId ? { warehouseId } : {};
536
537    const twoWeeksAgo = new Date();
538    twoWeeksAgo.setDate(twoWeeksAgo.getDate() - 14);
539
540    const [inventory, transactions] = await Promise.all([
541      prisma.inventory.findMany({
542        where: inventoryWhere,
543        include: { product: true, warehouse: true }
544      }),
545      prisma.transaction.findMany({
546        where: {
547          type: 'OUTBOUND',
548          timestamp: { gte: twoWeeksAgo },
549          ...(warehouseId && { warehouseId })
550        }
551      })
552    ]);
553
554    // Compute empirical daily consumption rate per SKU
555    const consumptionMap = new Map<string, number>();
556    transactions.forEach(tx => {
557      const current = consumptionMap.get(tx.productId) || 0;
558      consumptionMap.set(tx.productId, current + Math.abs(tx.quantity));
559    });
560
561    const predictions = inventory.map(item => {
562      const unitsConsumed14d = consumptionMap.get(item.productId) || 0;
563      const dailyVelocity = Number((unitsConsumed14d / 14).toFixed(2));
564      const daysRemaining = dailyVelocity > 0
565        ? Math.floor(item.quantity / dailyVelocity)
566        : 999;
567
568      let urgency: 'CRITICAL' | 'WARNING' | 'HEALTHY' = 'HEALTHY';
569      if (daysRemaining <= 3 || item.quantity <= 10) urgency = 'CRITICAL';
570      else if (daysRemaining <= 7 || item.quantity <= 30) urgency = 'WARNING';
571
572      // 14-day target safety buffer formula
573      const targetSafetyBuffer = Math.max(25, Math.ceil(dailyVelocity * 14));
574      const recommendedReorder = Math.max(0, targetSafetyBuffer - item.quantity);
575
576      return {
577        id: item.id,
578        sku: item.product.sku,
579        name: item.product.name,
580        price: Number(item.product.price),
581        warehouseCode: item.warehouse.code,
582        warehouseName: item.warehouse.name,
583        currentStock: item.quantity,
584        dailyVelocity,
585        daysRemaining: daysRemaining > 365 ? '365+' : daysRemaining,
586        urgency,
587        recommendedReorder
588      };
589    });
590
591    predictions.sort((a, b) => {
592      const rank = { CRITICAL: 0, WARNING: 1, HEALTHY: 2 };

16function playAcousticFeedback(type: 'scan' | 'success' | 'error') {
17  try {
18    const AudioCtx = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
19    if (!AudioCtx) return;
20    const ctx = new AudioCtx();
21    const osc = ctx.createOscillator();
22    const gain = ctx.createGain();
23    osc.connect(gain);
24    gain.connect(ctx.destination);
25
26    if (type === 'scan') {
27      osc.type = 'sine';
28      osc.frequency.setValueAtTime(1046.5, ctx.currentTime); // C6 tone
29      gain.gain.setValueAtTime(0.08, ctx.currentTime);
30      gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.06);
31      osc.start();
32      osc.stop(ctx.currentTime + 0.06);
33    } else if (type === 'success') {
34      osc.type = 'triangle';
35      osc.frequency.setValueAtTime(587.33, ctx.currentTime); // D5
36      osc.frequency.setValueAtTime(880, ctx.currentTime + 0.07); // A5
37      gain.gain.setValueAtTime(0.12, ctx.currentTime);
38      gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.22);
39      osc.start();
40      osc.stop(ctx.currentTime + 0.22);
41    } else if (type === 'error') {
42      osc.type = 'sawtooth';
43      osc.frequency.setValueAtTime(220, ctx.currentTime); // A3
44      osc.frequency.setValueAtTime(164.81, ctx.currentTime + 0.1); // E3
45      gain.gain.setValueAtTime(0.14, ctx.currentTime);
46      gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.28);
47      osc.start();
48      osc.stop(ctx.currentTime + 0.28);
49    }
50  } catch {
51    // Graceful fallback if autoplay policy or browser audio restrictions apply
52  }
